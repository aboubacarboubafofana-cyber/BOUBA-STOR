/* PATCH - Bouton Ajouter au panier en bas du bouton X - BOUBA STOR */
(function(){
  function addPanierButton(){
    document.querySelectorAll('#grid .card').forEach(card=>{
      if(card.querySelector('.btn-panier-bottom')) return; // déjà ajouté
      const pid=card.dataset.pid || card.getAttribute('data-pid') || (card.querySelector('[onclick*="addCart"]')?.getAttribute('onclick')?.match(/'([^']+)'/)?.[1]);
      if(!pid) return;
      
      // Trouve le bouton X
      const allBtns=[...card.querySelectorAll('button')];
      const xBtn=allBtns.find(b=>b.textContent.trim()==='X' || b.textContent.includes(' X ') || /^X$/.test(b.textContent.trim()));
      const container = xBtn ? xBtn.parentElement : card.querySelector('.info');
      if(!container) return;
      
      // Crée le bouton Ajouter au panier en bas de X
      const btn=document.createElement('button');
      btn.className='btn-panier-bottom';
      btn.style.cssText='width:100%;margin-top:8px;background:linear-gradient(135deg,#d4af37,#b8860b);color:#000;border:0;padding:12px;border-radius:10px;font-weight:900;font-size:13px;display:flex;align-items:center;justify-content:center;gap:6px;cursor:pointer;box-shadow:0 2px 10px rgba(212,175,55,.3)';
      btn.innerHTML='🛒 Ajouter au panier';
      btn.onclick=(e)=>{
        e.stopPropagation();
        if(typeof addCart==='function') addCart(pid);
        else {
          // fallback si addCart pas chargé
          let cart=JSON.parse(localStorage.getItem('b_cart')||'[]');
          const c=cart.find(x=>x.id===pid);
          if(c) c.qty++; else cart.push({id:pid,qty:1});
          localStorage.setItem('b_cart',JSON.stringify(cart));
          if(typeof renderCart==='function') renderCart();
        }
        btn.innerHTML='✅ Ajouté !';
        btn.style.background='#25D366';
        btn.style.color='#fff';
        if(navigator.vibrate) navigator.vibrate(50);
        setTimeout(()=>{ btn.innerHTML='🛒 Ajouter au panier'; btn.style.background=''; btn.style.color=''; btn.style.cssText+='background:linear-gradient(135deg,#d4af37,#b8860b);color:#000;'; },1500);
      };
      
      // Insère après le container de X, ou en bas de info
      if(xBtn && xBtn.parentElement.nextSibling){
        xBtn.parentElement.parentElement.appendChild(btn);
      } else {
        // si X est dans une grille 2 colonnes, on l'ajoute en bas de la carte
        const info=card.querySelector('.info') || card;
        info.appendChild(btn);
      }
    });
  }
  
  // Lance au chargement et à chaque renderStore
  const origRender=window.renderStore;
  if(origRender){
    window.renderStore=function(){ origRender(); setTimeout(addPanierButton,100); };
  }
  setInterval(addPanierButton,1000);
  document.addEventListener('DOMContentLoaded', addPanierButton);
  
  // CSS pour que le bouton X et le panier soient bien placés
  const style=document.createElement('style');
  style.textContent=`
    .btn-panier-bottom{animation:fadeIn .3s}
    @keyframes fadeIn{from{opacity:0;transform:translateY(10px)}to{opacity:1;transform:translateY(0)}}
    .btn-panier-bottom:active{transform:scale(.96)}
  `;
  document.head.appendChild(style);
})();
